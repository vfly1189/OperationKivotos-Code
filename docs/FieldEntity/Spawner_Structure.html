Operation Kivotos · 필드 엔티티 관리 · 스포너 구조

한 클래스의 네 가지 일을 네 층으로 나눈다

방법 1(손 배치 + ID 표)을 채택했으므로 데이터 형식은 그대로 두고 런타임만 다시 짠다. MonsterSpawner가 떠안던 저작 · 데이터 조인 · 상태 · 타이머를 층별로 떼어 내고, 포인트 하나의 상태를 명시적인 상태 기계로 만든다. 그러면 포인트마다 돌던 비동기 작업이 사라지고 S1~S3이 구조적으로 없어진다.

1. 전 / 후

왼쪽이 지금, 오른쪽이 제안. 아래층(MonsterFactory · Pool · 몬스터 생애)은 Phase 1 결과 그대로
그림 1 · SpawnerData(JSON) ↔ DataManager ↔ MonsterSpawner 구조의 전과 후
층누가하는 일하지 않는 일
데이터SpawnerData.json · DataManager표 로드 (형식 유지)—
저작MonsterSpawnerid · 자식 포인트 보관, 기즈모, Build(data) → Record스폰 · 상태 · 타이머 · 이벤트 구독
런타임SpawnerRecord · SpawnSlot포인트별 상태를 데이터로 보관스스로 움직이지 않음 (MonoBehaviour 아님)
시스템SpawnerSystem틱 · 전이 · 예산 · 사망 연결 · 씬 정리누가 켜는지 모름 (정책과 무관)
정책Sector (→ 거리 · 청크)켜질 스포너 id 결정몬스터 · 풀 · 팩토리를 모름

2. 씬 로드 — 한 번만 조인하고 검증한다

지금은 "켜질 때마다" 조인해서 오류가 늦게, 조용히 드러난다
그림 2 · 씬 로드 시 Record가 만들어지는 흐름

첫 스폰 대기(아래 5절 결정)에 따라 Init에서 슬롯의 ReadyAt을 0(켜지는 즉시)으로 둘지, 활성화 시각 + initialDelay로 둘지가 갈린다.

3. 슬롯 상태 기계 — 이 문서의 핵심

포인트 하나 = 슬롯 하나. 상태는 4개, 전이는 누가 일으키는지가 정해져 있다
그림 3 · SpawnSlot 상태 전이 (실선 = 평소, 점선 = Deactivate)
전이일으키는 것시스템이 하는 일
Waiting → SpawningOnUpdate 틱조건 3개 확인 → Generation++ → 팩토리 호출 (씬 CTS 하나 전달)
Spawning → Alive팩토리 반환반환 시 받아 둔 번호와 슬롯의 Generation이 같을 때만 채택. 다르면 Resource.Destroy (늦게 도착한 결과)
Alive → DyingMonsterStat.OnMonsterDead슬롯 상태만 바꿈. 점유는 유지 → 이 동안 같은 포인트에 새 몹 없음 (S3)
Dying → WaitingOnDespawned딕셔너리에서 연결 제거, ReadyAt = now + RespawnSec
Spawning → WaitingDeactivateGeneration++ 만으로 진행 중인 요청을 무효화. CTS를 포인트마다 둘 필요 없음 (S2)
Alive → WaitingDeactivate연결을 먼저 끊고 Resource.Destroy (→ Pool.Push → OnDespawn). 사망 경로가 아니므로 OnDespawned는 안 불림

개발 빌드에서 검사해 두면, 거리 정책을 붙일 때 "몬스터 층이 깨끗하다"는 증거가 된다
#불변식막는 것

I2슬롯 하나에 인스턴스는 최대 1개, 인스턴스 하나는 슬롯 최대 1개S3 한 포인트 2마리

I4HP바 수 = 활성 몬스터 수 (Phase 1 조건 재사용)HP바 잔존

6. 코드 뼈대

이름·시그니처는 제안. 실제 구현에서 바뀔 수 있다
// ── 런타임 데이터 — MonoBehaviour 아님. 합성 N개도 같은 형태 (청크 스케일 테스트)
public enum SlotState { Waiting, Spawning, Alive, Dying }

public sealed class SpawnSlot
{
    public readonly int MonsterId;  public readonly float RespawnSec;
    public readonly Vector3 Position; public readonly Quaternion Rotation;
    public SlotState State;  public float ReadyAt;  public int Generation;
    public BaseMonsterController Instance;
}

public sealed class SpawnerRecord
{
    public readonly int Id;  public readonly Vector3 Center;  public readonly float Radius;  // 정책이 읽는 전부
    public readonly SpawnSlot[] Slots;
    public bool IsActive;
}

// ── 저작 — 로직 없음. 씬 로드 때 한 번 Record로 바뀌고 끝
public class MonsterSpawner : MonoBehaviour
{
    [SerializeField] int _spawnerId;                          // _mapId 삭제 → 맵에서 한 번
    public int SpawnerId => _spawnerId;
    public SpawnerRecord Build(SpawnerData data);            // 자식 좌표 + 표 조인 + 검증 경고
    void OnDrawGizmos();                                  // 기존 그대로
}

// ── 시스템 — 씬 수명. Managers.Update에서 OnUpdate, 씬 전환에서 Clear
public class SpawnerSystem
{
    readonly Dictionary<int, SpawnerRecord> _records;
    readonly Dictionary<BaseMonsterController, SpawnSlot> _owner;   // 사망 연결 (람다 구독 대신)
    CancellationTokenSource _sceneCts;                                // 씬에 하나
    public int SpawnBudgetPerFrame = int.MaxValue;                  // 공통 스위치 (기본 무제한)

    public void Init(int mapId, IEnumerable<SpawnerRecord> records);
    public void Activate(int spawnerId);
    public void Deactivate(int spawnerId);   // 그림 3의 점선 전이
    public void OnUpdate();                  // 활성 레코드의 기한 된 Waiting → 예산 안에서 스폰
    public void Clear();

    async UniTaskVoid SpawnAsync(SpawnSlot slot, int gen);   // 반환 시 gen != slot.Generation 이면 회수
    void HandleDead(BaseMonsterController m);                // Alive → Dying
    void HandleDespawned(BaseMonsterController m);           // Dying → Waiting
}

// ── 정책 (지금은 Sector) — 스포너 참조 대신 id만
public class Sector : MonoBehaviour
{
    int[] _spawnerIds;                                     // Start에서 자식 MonsterSpawner의 id 수집
    public void ActivateSector()   { foreach (var id in _spawnerIds) Managers.Spawner.Activate(id); }
    public void DeactivateSector() { foreach (var id in _spawnerIds) Managers.Spawner.Deactivate(id); }
}
몬스터 쪽 작은 변경 하나 BaseMonsterController.OnDespawned를 Action → Action<BaseMonsterController>로 바꾼다. 그래야 시스템이 핸들러 하나로 "누가 끝났는지" 알 수 있고, 생애마다 람다를 만들지 않는다. 두 이벤트(OnMonsterDead, OnDespawned)는 이미 매 스폰에 초기화되므로(SetStat · OnSpawn) 팩토리 반환 뒤 구독해도 안전하다.

정할 것 — 첫 스폰 대기 지금 JSON의 delay(363개 전부 5.0초)는 첫 스폰 대기와 리스폰 시간을 겸한다. 구역에 들어가면 5초 동안 몹이 없다. 두 정책에 똑같이 걸리지만 "들어갔는데 몹이 없다"는 체감 지표를 흐린다. 제안: delay → respawnSec로 의미를 고정하고, 첫 활성화는 즉시 스폰 (ReadyAt = 0). 필요하면 initialDelay(기본 0)를 추가.
